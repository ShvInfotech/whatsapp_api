Dim url As String = "http://93.127.198.184/whatsapp_api/api/send?number=" & strMobile & "&type=text&message=" & Uri.EscapeDataString(strMsg) & "&instance_id=" & strInstance & "&access_token=" & strToken
Dim req As HttpWebRequest = WebRequest.Create(url)
Dim res As HttpWebResponse = req.GetResponse()
